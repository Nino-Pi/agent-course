#!/usr/bin/env python3
"""Offline teaching lab: persisted preferences, fresh readers, scoped projection.

This is a small teaching design, not Maka code. No LLM is called and no answer
language is generated. Sequential JSON writes provide no crash/transaction
guarantee. All parent-controlled output lives below .workspace/learning-labs.
"""
from __future__ import annotations

import argparse
import hashlib
import json
import os
from pathlib import Path
import subprocess
import sys


ROOT = Path(__file__).resolve().parents[5]
LAB_ROOT = (ROOT / '.workspace/learning-labs').resolve()
REQUEST = '解释 formatDuration 的测试结果。'


def checked_path(raw: str) -> Path:
    path = Path(raw).resolve()
    if path == LAB_ROOT or LAB_ROOT not in path.parents:
        raise ValueError('Output must be a child of .workspace/learning-labs')
    return path


def worker(action: str, store: Path, workspace: str, read_enabled: bool) -> dict:
    if action == 'save-global':
        data = {'global': 'zh', 'workspaces': {}}
        store.write_text(json.dumps(data, ensure_ascii=False, indent=2) + '\n')
        return {'pid': os.getpid(), 'action': action, 'saved': data}
    if action == 'save-override':
        data = json.loads(store.read_text())
        data['workspaces'][workspace] = 'en'
        store.write_text(json.dumps(data, ensure_ascii=False, indent=2) + '\n')
        return {'pid': os.getpid(), 'action': action, 'saved': data}
    if not read_enabled:
        # No store read is needed for this branch. The file still exists.
        return {'pid': os.getpid(), 'action': action, 'workspace': workspace,
                'current_request': REQUEST, 'background': [], 'store_read': False}
    data = json.loads(store.read_text())
    # The deliberately broken variant ignores the workspace exception.
    language = (data['global'] if action == 'read-global-only'
                else data['workspaces'].get(workspace, data['global']))
    return {'pid': os.getpid(), 'action': action, 'workspace': workspace,
            'current_request': REQUEST, 'store_read': True,
            'background': [{'kind': 'explanation-language', 'value': language,
                            'source': 'saved-preference', 'authority': 'background'}]}


def child(action: str, store: Path, workspace: str = 'A', enabled: bool = True) -> dict:
    command = [sys.executable, str(Path(__file__).resolve()), '--worker', action,
               '--store', str(store), '--workspace', workspace]
    if not enabled:
        command.append('--read-disabled')
    completed = subprocess.run(command, check=True, capture_output=True, text=True, timeout=15)
    return json.loads(completed.stdout)


def run(output: Path) -> dict:
    # Refuse old runs so generated evidence cannot silently replace earlier data.
    output.mkdir(parents=True, exist_ok=False)
    store = output / 'preferences.json'
    saved = child('save-global', store)
    first = child('read', store)
    updated = child('save-override', store)
    project_a = child('read', store, 'A')
    project_b = child('read', store, 'B')
    before_disabled = hashlib.sha256(store.read_bytes()).hexdigest()
    disabled = child('read', store, 'A', enabled=False)
    after_disabled = hashlib.sha256(store.read_bytes()).hexdigest()
    broken = child('read-global-only', store, 'A')

    checks = {
        'fresh_reader_recovers_global_preference': first['background'][0]['value'] == 'zh',
        'workspace_a_selects_override': project_a['background'][0]['value'] == 'en',
        'workspace_b_keeps_global_preference': project_b['background'][0]['value'] == 'zh',
        'disabled_read_has_no_background': disabled['background'] == [] and not disabled['store_read'],
        'disabled_read_preserves_store': before_disabled == after_disabled,
        'new_request_does_not_repeat_preference': all(
            row['current_request'] == REQUEST and '中文' not in row['current_request']
            and '英文' not in row['current_request']
            for row in [first, project_a, project_b, disabled, broken]
        ),
        'negative_control_exposes_wrong_scope': broken['background'][0]['value'] == 'zh'
            and broken['background'] != project_a['background'],
    }
    report = {
        'result': 'PASS' if all(checks.values()) else 'FAIL',
        'design': 'teaching-only; sequential JSON writes; each action in a fresh subprocess',
        'model_called': False,
        'maka_called': False,
        'proves': 'cross-process file retention and deterministic scoped background selection',
        'does_not_prove': ['model compliance', 'Maka wiring', 'crash durability', 'concurrent writes'],
        'checks': checks,
        'trace': [saved, first, updated, project_a, project_b, disabled, broken],
    }
    (output / 'verification.json').write_text(json.dumps(report, ensure_ascii=False, indent=2) + '\n')
    if not all(checks.values()):
        raise AssertionError(json.dumps(checks, ensure_ascii=False))
    return report


def main() -> None:
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument('--output')
    parser.add_argument('--worker', choices=['save-global', 'save-override', 'read', 'read-global-only'])
    parser.add_argument('--store')
    parser.add_argument('--workspace', default='A')
    parser.add_argument('--read-disabled', action='store_true')
    args = parser.parse_args()
    if args.worker:
        if not args.store:
            parser.error('--worker requires --store')
        result = worker(args.worker, checked_path(args.store), args.workspace, not args.read_disabled)
    else:
        if not args.output:
            parser.error('--output is required; choose a new directory for every run')
        result = run(checked_path(args.output))
    print(json.dumps(result, ensure_ascii=False, indent=2))


if __name__ == '__main__':
    main()
