#!/usr/bin/env python3
"""Offline control-flow lab using unmodified mini-SWE-agent source.

The response sequence is scripted. It cannot discover or adapt to another bug.
All generated files belong in .workspace/learning-labs; existing runs are retained.
"""

import argparse
import json
import os
from pathlib import Path
import shlex
import shutil
import signal
import subprocess
import sys
from types import SimpleNamespace

ASSETS = Path(__file__).resolve().parent
ROOT = ASSETS.parents[4]
TASK = "核对 README 的测试命令，修复时长函数的缺陷，运行测试，检查 diff，用中文总结。"


def command(args, cwd=None, check=True):
    return subprocess.run(args, cwd=cwd, text=True, capture_output=True, check=check)


def write_json(path, value):
    path.write_text(json.dumps(value, ensure_ascii=False, indent=2) + "\n")


def prepare_source(source, output):
    required = ["agents/default.py", "environments/local.py", "models/test_models.py"]
    for module in required:
        if not (source / "src/minisweagent" / module).is_file():
            raise SystemExit(f"Missing mini source module: {module}")
    if command(["git", "-C", str(source), "status", "--porcelain", "--untracked-files=no"]).stdout:
        raise SystemExit("The mini source has tracked modifications; use a clean source directory.")
    sys.dont_write_bytecode = True
    sys.path.insert(0, str(source / "src"))
    os.environ.update({
        "MSWEA_GLOBAL_CONFIG_DIR": str(output / "mini-config"),
        "MSWEA_SILENT_STARTUP": "1",
        "MSWEA_GLOBAL_COST_LIMIT": "0",
        "MSWEA_GLOBAL_CALL_LIMIT": "0",
    })


def create_fixture(output):
    fixture = output / "repo"
    shutil.copytree(ASSETS / "fixture", fixture)
    command(["git", "init", "-q"], cwd=fixture)
    command(["git", "add", "README.md", "package.json", "src", "test"], cwd=fixture)
    command(["git", "-c", "core.hooksPath=/dev/null", "-c", "commit.gpgsign=false",
             "-c", "user.name=Learning Lab", "-c", "user.email=lab@example.invalid",
             "commit", "-qm", "Initial intentionally broken teaching fixture"], cwd=fixture)
    return fixture


def python_action(code):
    return f"{shlex.quote(sys.executable)} -c {shlex.quote(code)}"


def repair_readme():
    return "from pathlib import Path; p=Path('README.md'); p.write_text(p.read_text().replace('npm run check', 'npm test'))"


def make_agent(output, plan):
    from minisweagent.agents.default import DefaultAgent
    from minisweagent.environments.local import LocalEnvironment
    from minisweagent.models.test_models import DeterministicToolcallModel, make_toolcall_output
    from minisweagent.models.utils.actions_toolcall import parse_toolcall_actions

    responses = []
    for index, (label, shell_command) in enumerate(plan, start=1):
        call = {"id": f"call_{index}", "type": "function", "function": {
            "name": "bash", "arguments": json.dumps({"command": shell_command})}}
        parsed = parse_toolcall_actions(
            [SimpleNamespace(id=call["id"], function=SimpleNamespace(**call["function"]))],
            format_error_template="{{ error }}",
        )
        responses.append(make_toolcall_output(label, [call], parsed))
    model = DeterministicToolcallModel(outputs=responses)
    env = LocalEnvironment(cwd=str(output / "repo"), timeout=15,
                           env={"npm_config_cache": str(output / "npm-cache"),
                                "npm_config_update_notifier": "false"})
    return DefaultAgent(model, env, system_template="Follow the task with the bash tool.",
                        instance_template="{{ task }}", cost_limit=0, step_limit=len(plan) + 1,
                        output_path=output / "trajectory.json")


def run_repair(output):
    plan = [
        ("读取 README 和测试入口", "cat README.md package.json"),
        ("照 README 执行，观察不存在的脚本", "npm run check"),
        ("定位函数及测试断言", "cat src/duration.js test/duration.test.js"),
        ("使用真实测试入口，观察函数失败", "npm test"),
        ("修正文档和取余基数", python_action(repair_readme() +
            "; p=Path('src/duration.js'); p.write_text(p.read_text().replace('seconds % 100', 'seconds % 60'))")),
        ("再次运行测试", "npm test"),
        ("查看修改范围", "git diff -- README.md src/duration.js"),
        ("提交已观察到的结果", "printf '%s\\n' COMPLETE_TASK_AND_SUBMIT_FINAL_OUTPUT '已修正测试命令和秒数取余，4 个测试通过；diff 仅含两个目标文件。'"),
    ]
    agent = make_agent(output, plan)
    result = agent.run(TASK)
    observations = [message for message in agent.messages if message.get("role") == "tool"]
    returncodes = [message["extra"]["returncode"] for message in observations]
    expected = [0, 1, 0, 1, 0, 0, 0]
    if returncodes != expected:
        raise AssertionError(f"Expected observed return codes {expected}; found {returncodes}")
    if "Missing script" not in observations[1]["extra"]["raw_output"]:
        raise AssertionError("The README failure was not the missing-script case")
    before = observations[3]["extra"]["raw_output"]
    after = observations[5]["extra"]["raw_output"]
    if "fail 2" not in before or "pass 4" not in after or "fail 0" not in after:
        raise AssertionError("Expected two failing tests followed by four passing tests")
    diff = command(["git", "diff", "--", "README.md", "src/duration.js"], cwd=output / "repo").stdout
    changed = command(["git", "diff", "--name-only"], cwd=output / "repo").stdout.splitlines()
    if changed != ["README.md", "src/duration.js"] or result["exit_status"] != "Submitted":
        raise AssertionError("Unexpected changed-file scope or agent exit")
    (output / "diff.patch").write_text(diff)
    write_json(output / "command-trace.json", [
        {"step": index, "label": label, "command": shell_command,
         "observation": observations[index - 1]["extra"] if index <= len(observations) else None}
        for index, (label, shell_command) in enumerate(plan, 1)
    ])
    verification = {"scenario": "repair", "source": "mini-swe-agent", "model": "DeterministicToolcallModel",
                    "scripted_not_model_evaluation": True, "model_calls": agent.n_calls,
                    "synthetic_cost_units": agent.cost, "tool_observations": len(observations),
                    "observed_returncodes": returncodes, "tests_before": {"pass": 2, "fail": 2},
                    "tests_after": {"pass": 4, "fail": 0}, "changed_files": changed,
                    "exit_status": result["exit_status"], "acceptance": "PASS"}
    write_json(output / "verification.json", verification)
    print(json.dumps(verification, ensure_ascii=False, indent=2))


def crash_worker(output):
    # The shell writes only this generated README, then kills this specific child agent.
    # It exits immediately afterwards; no sleeping or detached tool is created.
    effect = python_action(repair_readme() +
                           "; Path('../effect-written.txt').write_text('README changed before owner exit\\n')")
    plan = [("读取修改前的 README", "cat README.md"),
            ("写入后在 observation 返回前注入崩溃", f"{effect} && kill -KILL {os.getpid()}")]
    make_agent(output, plan).run(TASK)
    raise AssertionError("Crash worker unexpectedly survived")


def run_crash(source, output):
    if os.name != "posix":
        raise SystemExit("The injected SIGKILL scenario currently requires a POSIX environment.")
    worker = command([sys.executable, str(Path(__file__).resolve()), "crash-worker",
                      "--source", str(source), "--output", str(output)], check=False)
    (output / "worker.stdout.log").write_text(worker.stdout)
    (output / "worker.stderr.log").write_text(worker.stderr)
    if worker.returncode != -signal.SIGKILL:
        raise AssertionError(f"Expected SIGKILL (-9); found {worker.returncode}")
    trajectory = json.loads((output / "trajectory.json").read_text())
    roles = [message["role"] for message in trajectory["messages"]]
    old_observation = trajectory["messages"][-1]["extra"]["raw_output"]
    readme = (output / "repo" / "README.md").read_text()
    if roles != ["system", "user", "assistant", "tool"] or "npm run check" not in old_observation:
        raise AssertionError("The saved trajectory was not the first completed read step")
    if "npm test" not in readme or not (output / "effect-written.txt").exists():
        raise AssertionError("The intended filesystem effect did not precede the crash")
    diff = command(["git", "diff", "--", "README.md"], cwd=output / "repo").stdout
    (output / "diff.patch").write_text(diff)
    verification = {"scenario": "crash", "source": "mini-swe-agent", "worker_returncode": worker.returncode,
                    "saved_roles": roles, "saved_model_calls": trajectory["info"]["model_stats"]["api_calls"],
                    "saved_observation_has_old_command": True, "filesystem_has_corrected_command": True,
                    "saved_write_action_in_messages": False, "saved_write_observation": False,
                    "saved_exit": False, "automatic_recovery_tested": False, "acceptance": "PASS"}
    write_json(output / "verification.json", verification)
    print(json.dumps(verification, ensure_ascii=False, indent=2))


def main():
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument("scenario", choices=["prepare", "repair", "crash", "crash-worker"])
    parser.add_argument("--source", type=Path,
                        default=ROOT / "repositories/upstreams/mini-swe-agent")
    parser.add_argument("--output", required=True, type=Path)
    args = parser.parse_args()
    source, output = args.source.resolve(), args.output.resolve()
    generated_root = ROOT / ".workspace/learning-labs"
    if not output.is_relative_to(generated_root) or output == generated_root:
        raise SystemExit("Output must be a new subdirectory of .workspace/learning-labs")
    if args.scenario == "crash-worker":
        prepare_source(source, output)
        crash_worker(output)
        return
    if output.exists():
        raise SystemExit("Output already exists. Choose a new directory to preserve earlier runs.")
    for executable in ["git", "node", "npm"]:
        if not shutil.which(executable):
            raise SystemExit(f"Missing required executable: {executable}")
    output.mkdir(parents=True)
    prepare_source(source, output)
    create_fixture(output)
    write_json(output / "versions.json", {"source": os.path.relpath(source, ROOT), "python": sys.version.split()[0],
               "node": command(["node", "--version"]).stdout.strip(),
               "npm": command(["npm", "--version"]).stdout.strip(), "platform": sys.platform})
    if args.scenario == "prepare":
        print("Prepared intentionally broken fixture; no agent was run.")
    elif args.scenario == "repair":
        run_repair(output)
    else:
        run_crash(source, output)


if __name__ == "__main__":
    main()
